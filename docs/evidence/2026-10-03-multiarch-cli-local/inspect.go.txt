package main

import (
 "context"
 "crypto/sha256"
 "encoding/json"
 "fmt"
 "os"
 "path/filepath"
 "time"
 "github.com/NVIDIA/nke-recipes/pkg/nkd/delivery"
)

func main() {
 ctx, cancel := context.WithTimeout(context.Background(), time.Minute)
 defer cancel()
 run := ".nkd/runs/nkd-cli-dual-20261003-02"
 root := ".nkd/evidence-cli/diagnostic-root"
 var objects []delivery.ObjectInput
 for _, arch := range []string{"amd64", "arm64"} {
  name := "nkectl-linux-"+arch
  bundle, result, err := delivery.InspectBinaryBundle(ctx, filepath.Join(run, "packages", name))
  must(err)
  if bundle.OS != "linux" || bundle.Arch != arch || bundle.Purpose != name { panic("wrong output binding") }
  objects = append(objects, delivery.ObjectInput{Name:name, Kind:"binary-bundle", Path:result.LayoutPath, ArtifactType:delivery.MediaBinaryBundle, SemanticDigest:result.SemanticDigest, BlobDigest:result.RootDigest, OS:"linux", Arch:arch, Layout:true})
  raw, err := json.Marshal(struct { Name string; Bundle delivery.BinaryBundle; Result delivery.Result }{name,bundle,result})
  must(err)
  fmt.Println(string(raw))
 }
 result, err := delivery.BuildPublicationRoot(ctx,root,delivery.PublicationRequest{Objects:objects})
 must(err)
 fmt.Printf("diagnostic local root: %s\n",result.RootDigest)
 for _, arch := range []string{"amd64","arm64"} {
  name := "nkectl-linux-"+arch
  out := filepath.Join(".nkd/evidence-cli/extracted",name)
  bundle, err := delivery.ExtractBinaryBundle(ctx,root,name,out)
  must(err)
  recovered, err := os.ReadFile(filepath.Join(out,"nkectl")); must(err)
  original, err := os.ReadFile(filepath.Join(run,name,"out/nkectl/nkectl")); must(err)
  if sha256.Sum256(recovered) != sha256.Sum256(original) || fmt.Sprintf("sha256:%x",sha256.Sum256(recovered)) != bundle.Entries[0].SHA256 { panic("extracted bytes changed") }
  fmt.Printf("%s extracted payload sha256:%x; exact producer bytes verified\n",name,sha256.Sum256(recovered))
 }
}
func must(err error) { if err != nil { panic(err) } }
