// Diagnostic replay only. This creates raw preparation evidence, not a signed
// publication or a production-authorized Release. Run from the repository root.
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"time"

	"github.com/NVIDIA/nke-recipes/pkg/nkd/delivery"
	"github.com/NVIDIA/nke-recipes/pkg/nkd/sbom"
)

func main() {
	if len(os.Args) != 4 {
		panic("usage: replay <CLI-worktree> <new-layout-path> <loopback-registry-repository>")
	}
	if !strings.HasPrefix(os.Args[3], "127.0.0.1:") {
		panic("diagnostic replay requires a loopback registry")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Minute)
	defer cancel()
	source, destination, repository := os.Args[1], os.Args[2], os.Args[3]
	records := filepath.Join(source, "docs/evidence/2026-10-03-multiarch-cli-local")
	// These two recorded identities are descriptive fixture inputs. This helper
	// does not authenticate the Run or validate the CLI branch's newer Plan schema.
	planBytes, err := os.ReadFile(filepath.Join(records, "plan.json"))
	must(err)
	var plan struct {
		DefinitionDigest string `json:"definitionDigest"`
		EnvironmentDigest string `json:"environmentDigest"`
	}
	must(json.Unmarshal(planBytes, &plan))
	request := delivery.PublicationRequest{Release: &delivery.ReleaseMetadata{
		DefinitionDigest: plan.DefinitionDigest, EnvironmentDigest: plan.EnvironmentDigest,
	}}
	var outputs []delivery.Output
	for _, arch := range []string{"amd64", "arm64"} {
		name := "nkectl-linux-" + arch
		layout := filepath.Join(source, ".nkd/runs/nkd-cli-dual-20261003-02/packages", name)
		bundle, result, err := delivery.InspectBinaryBundle(ctx, layout)
		must(err)
		if bundle.OS != "linux" || bundle.Arch != arch || len(bundle.Entries) != 1 || bundle.Entries[0].Name != "nkectl" {
			panic("unexpected binary bundle platform or members")
		}
		descriptor, err := delivery.PayloadDescriptor(ctx, layout)
		must(err)
		raw, err := os.ReadFile(filepath.Join(records, arch+"-sbom.json"))
		must(err)
		var bom sbom.BOM
		must(json.Unmarshal(raw, &bom))
		must(bom.Validate())
		buildDigest := ""
		for _, property := range bom.Metadata.Props {
			if property.Name == "nkd:buildDigest" {
				buildDigest = property.Value
			}
		}
		subject := bundle.Entries[0].SHA256
		if len(bom.Metadata.Component.Hashes) != 1 || "sha256:"+bom.Metadata.Component.Hashes[0].Content != subject {
			panic("SBOM does not name the exact packaged executable")
		}
		bound, err := sbom.ForSubject(&bom, subject, "build-output", subject)
		must(err)
		evidence, err := bound.Marshal()
		must(err)
		semantic, err := bound.Digest()
		must(err)
		outputs = append(outputs, delivery.Output{Name: name, Kind: "executable", Digest: descriptor.Digest, BuildDigest: buildDigest, Components: []string{"nkectl"}})
		request.Objects = append(request.Objects,
			delivery.ObjectInput{Name: name, Kind: "binary-bundle", Path: layout, Layout: true,
				ArtifactType: delivery.MediaBinaryBundle, SemanticDigest: result.SemanticDigest, BlobDigest: result.RootDigest,
				Output: &delivery.OutputBinding{Repository: name, Format: "executable", BuildDigest: buildDigest,
					Components: []string{"nkectl"}, BuildSubjects: []delivery.BuildSubject{{Component: "nkectl", Digest: subject}}}},
			delivery.ObjectInput{Name: "sbom-" + strings.TrimPrefix(semantic, "sha256:"), Kind: "sbom",
				ArtifactType: "application/vnd.cyclonedx+json", LayerMediaType: "application/vnd.cyclonedx+json", Bytes: evidence, SemanticDigest: semantic,
				Evidence: &delivery.EvidenceBinding{Outputs: []string{name}, Component: "nkectl", Stage: "build-output", SubjectKind: "file", SubjectDigest: subject}},
		)
	}
	request.Release.ContentSubject, err = delivery.ContentDigest(outputs)
	must(err)
	result, err := delivery.BuildPublicationRoot(ctx, destination, request)
	must(err)
	publication, err := delivery.OpenPublication(ctx, destination, result.RootDigest)
	must(err)
	must(publication.Close())
	reference, err := delivery.Push(ctx, destination, repository, delivery.RegistryOptions{PlainHTTP: true, Anonymous: true})
	must(err)
	fmt.Println(reference)
}

func must(err error) {
	if err != nil {
		panic(err)
	}
}
